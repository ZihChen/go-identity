```go
// New configuration structure
type WorkerConfig struct {
    Concurrency     int
    QueuePriorities map[string]int
    TaskTimeout     time.Duration
    MaxRetries      int
    RetryDelay      func(int) time.Duration
}

// Environment-specific configurations
func getDevConfig() WorkerConfig {
    return WorkerConfig{
        Concurrency: 3,           // Reduced for memory constraints
        QueuePriorities: map[string]int{
            "default":  2,        // Lower priority allocation
            "critical": 3,
        },
        TaskTimeout: 15 * time.Second,  // Shorter timeout
        MaxRetries: 3,            // Fewer retries
        RetryDelay: func(n int) time.Duration {
            return time.Duration(n) * 2 * time.Second  // Linear backoff
        },
    }
}

func getProdConfig() WorkerConfig {
    return WorkerConfig{
        Concurrency: 20,          // Utilize dual-core fully
        QueuePriorities: map[string]int{
            "default":   8,
            "critical":  15,
            "bulk":      5,       // New bulk processing queue
        },
        TaskTimeout: 60 * time.Second,  // Extended timeout
        MaxRetries: 5,            // More resilient retries
        RetryDelay: func(n int) time.Duration {
            delay := time.Duration(n*n) * time.Second
            if delay > 5*time.Minute {
                return 5 * time.Minute  // Exponential with cap
            }
            return delay
        },
    }
}
```

#### 3.1 Environment-Aware Batching

```go
// New batch processing methods
const (
    TypeBulkSync = "bulk:sync"  // New task type for batching
)

func (q *QueueService) enqueueBatchTask(ctx context.Context, taskType string, tasks [][]byte) error {
    if q.cfg.App.Env == "dev" {
        // Process sequentially in dev environment
        for _, data := range tasks {
            if err := q.enqueueTask(ctx, taskType, data); err != nil {
                return err
            }
        }
        return nil
    }
    
    // Batch processing for production
    return q.enqueueBatchOptimized(ctx, taskType, tasks)
}

func (q *QueueService) enqueueBatchOptimized(ctx context.Context, taskType string, tasks [][]byte) error {
    // Create a single batch task with multiple payloads
    batchData, err := json.Marshal(map[string]interface{}{
        "type":  "batch",
        "tasks": tasks,
    })
    if err != nil {
        return err
    }
    
    return q.enqueueTask(ctx, TypeBulkSync, batchData)
}
```
