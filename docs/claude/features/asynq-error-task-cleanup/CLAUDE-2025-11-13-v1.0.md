# Asynq錯誤事件收集與Redis清理系統 (簡化版)

**版本**: v1.0  
**日期**: 2025-11-13  
**狀態**: 設計階段

## 概述

本文檔描述了Fat-Identity-Cat專案中Asynq錯誤事件收集流程的簡化設計與實現，專注於兩個核心功能：
1. 將Redis上執行失敗的事件錯誤訊息寫入資料庫做永久性儲存
2. 自動清理Redis上失敗事件的相關資料

## 目標

1. **錯誤事件持久化** - 將Redis上執行失敗的Asynq任務錯誤訊息永久存儲至MySQL資料庫
2. **智能錯誤分類** - 根據錯誤訊息自動分類錯誤類型，便於分析和監控
3. **Redis失敗資料清理** - 自動清理Redis中失敗任務相關的過期資料
4. **分佈式追蹤集成** - 與現有OpenTelemetry系統無縫集成
5. **Clean Architecture合規** - 完全符合專案現有的六角型架構模式

## 當前架構分析

### 現有技術棧
- **Asynq**: `hibiken/asynq v0.25.1` - Redis背景任務隊列
- **Redis**: 任務存儲和緩存層
- **MySQL**: 持久化存儲
- **Clean Architecture**: 六角型架構
- **OpenTelemetry**: 分佈式追蹤和監控

### 現有錯誤處理機制
1. **Queue Service** (`internal/infrastructure/queue/queue.go`): 
   - 完善的重試策略（指數退避，最多5次重試）
   - 任務超時設置（30秒）
   - 錯誤日誌記錄和追蹤集成

2. **Worker Handler** (`internal/adapter/inbound/handler/worker/worker_handler.go`):
   - 各類同步任務處理器（Merchant, Player, Manager, Level, Tag, Agent）
   - 詳細的錯誤日誌記錄
   - OpenTelemetry span追蹤

// internal/adapter/outbound/repository/failed_task_event/failed_task_event_repository.go
package failed_task_event

import (
    "context"
    "time"
    "gorm.io/gorm"
    "github.com/jvdiamondtech/ms-identity-cat/internal/domain/entity"
    "github.com/jvdiamondtech/ms-identity-cat/internal/domain/ports/outbound/repository"
    "github.com/jvdiamondtech/ms-identity-cat/internal/infrastructure/models"
)

type Repository struct {
    db *gorm.DB
}

func NewFailedTaskEventRepository(db *gorm.DB) repository.FailedTaskEventRepository {
    return &Repository{db: db}
}

func (r *Repository) Save(ctx context.Context, event *entity.FailedTaskEvent) error {
    model := &models.FailedTaskEvent{
        TaskID:       event.GetTaskID(),
        TaskType:     event.GetTaskType(),
        QueueName:    event.GetQueueName(),
        Payload:      event.GetPayload(),
        ErrorMessage: event.GetErrorMessage(),
        ErrorType:    event.GetErrorType(),
        RetryCount:   event.GetRetryCount(),
        FailedAt:     event.GetFailedAt(),
        TraceID:      event.GetTraceID(),
        SpanID:       event.GetSpanID(),
    }
    
    return r.db.WithContext(ctx).Create(model).Error
}

func (r *Repository) FindByTimeRange(ctx context.Context, startTime, endTime time.Time) ([]*entity.FailedTaskEvent, error) {
    var models []*models.FailedTaskEvent
    err := r.db.WithContext(ctx).
        Where("failed_at BETWEEN ? AND ?", startTime, endTime).
        Order("failed_at DESC").
        Find(&models).Error
    
    if err != nil {
        return nil, err
    }
    
    events := make([]*entity.FailedTaskEvent, len(models))
    for i, model := range models {
        events[i] = r.mapToDomainEntity(model)
    }
    
    return events, nil
}

func (r *Repository) FindByTaskType(ctx context.Context, taskType string, limit int) ([]*entity.FailedTaskEvent, error) {
    var models []*models.FailedTaskEvent
    err := r.db.WithContext(ctx).
        Where("task_type = ?", taskType).
        Order("failed_at DESC").
        Limit(limit).
        Find(&models).Error
    
    if err != nil {
        return nil, err
    }
    
    events := make([]*entity.FailedTaskEvent, len(models))
    for i, model := range models {
        events[i] = r.mapToDomainEntity(model)
    }
    
    return events, nil
}

func (r *Repository) DeleteOldEvents(ctx context.Context, beforeTime time.Time) (int64, error) {
    result := r.db.WithContext(ctx).
        Where("created_at < ?", beforeTime).
        Delete(&models.FailedTaskEvent{})
    return result.RowsAffected, result.Error
}

func (r *Repository) GetErrorStatistics(ctx context.Context, startTime, endTime time.Time) (map[string]int, error) {
    type errorStat struct {
        ErrorType string
        Count     int
    }
    
    var stats []errorStat
    err := r.db.WithContext(ctx).
        Model(&models.FailedTaskEvent{}).
        Select("error_type, COUNT(*) as count").
        Where("failed_at BETWEEN ? AND ?", startTime, endTime).
        Group("error_type").
        Scan(&stats).Error
    
    if err != nil {
        return nil, err
    }
    
    result := make(map[string]int)
    for _, stat := range stats {
        result[stat.ErrorType] = stat.Count
    }
    
    return result, nil
}

func (r *Repository) mapToDomainEntity(model *models.FailedTaskEvent) *entity.FailedTaskEvent {
    event := entity.NewFailedTaskEventWithTimes(
        model.TaskID,
        model.TaskType,
        model.QueueName,
        model.Payload,
        model.ErrorMessage,
        model.ErrorType,
        model.RetryCount,
        model.FailedAt,
        model.CreatedAt,
        model.UpdatedAt,
    )
    
    event.SetID(model.ID)
    
    if model.TraceID != nil {
        event.SetTraceID(*model.TraceID)
    }
    if model.SpanID != nil {
        event.SetSpanID(*model.SpanID)
    }
    if model.DeletedAt.Valid {
        deletedTime := model.DeletedAt.Time
        event.SetDeletedAt(&deletedTime)
    }
    
    return event
}

// internal/application/usecase/failed_task_event_usecase.go
package usecase

import (
    "context"
    "fmt"
    "strings"
    "time"
    
    "github.com/jvdiamondtech/ms-identity-cat/internal/domain/entity"
    "github.com/jvdiamondtech/ms-identity-cat/internal/domain/ports/inbound"
    "github.com/jvdiamondtech/ms-identity-cat/internal/domain/ports/outbound/repository"
    "github.com/jvdiamondtech/ms-identity-cat/internal/domain/ports/outbound/infrastructure"
    "go.opentelemetry.io/otel/attribute"
    "go.opentelemetry.io/otel/codes"
)

type FailedTaskEventUseCase struct {
    repo    repository.FailedTaskEventRepository
    logger  infrastructure.Logger
    tracing infrastructure.TracingService
}

func NewFailedTaskEventUseCase(
    repo repository.FailedTaskEventRepository,
    logger infrastructure.Logger,
    tracing infrastructure.TracingService,
) inbound.FailedTaskEventUseCase {
    return &FailedTaskEventUseCase{
        repo:    repo,
        logger:  logger,
        tracing: tracing,
    }
}

func (u *FailedTaskEventUseCase) RecordFailedTask(
    ctx context.Context,
    taskID, taskType, queueName, payload, errorMessage string,
    retryCount int,
    traceID, spanID *string,
) error {
    ctx, span := u.tracing.StartSpan(ctx, "FailedTaskEventUseCase.RecordFailedTask")
    defer u.tracing.SpanEnd(span)
    
    u.tracing.RecordSpanAttributes(span,
        attribute.String("task.id", taskID),
        attribute.String("task.type", taskType),
        attribute.String("queue.name", queueName),
        attribute.Int("retry.count", retryCount),
    )
    
    // 分類錯誤類型
    errorType := u.classifyError(errorMessage)
    
    event := entity.NewFailedTaskEvent(
        taskID, taskType, queueName, payload, errorMessage, errorType,
        retryCount, time.Now(),
    )
    
    // 設置追蹤信息
    if traceID != nil {
        event.SetTraceID(*traceID)
        u.tracing.RecordSpanAttributes(span, attribute.String("trace.id", *traceID))
    }
    if spanID != nil {
        event.SetSpanID(*spanID)
        u.tracing.RecordSpanAttributes(span, attribute.String("span.id", *spanID))
    }
    
    // 驗證實體
    if !event.IsValid() {
        err := fmt.Errorf("invalid failed task event data")
        u.tracing.RecordSpanError(span, err)
        u.tracing.RecordSpanStatus(span, codes.Error, "Invalid event data")
        return err
    }
    
    // 保存到資料庫
    if err := u.repo.Save(ctx, event); err != nil {
        u.tracing.RecordSpanError(span, err)
        u.tracing.RecordSpanStatus(span, codes.Error, fmt.Sprintf("Failed to save: %v", err))
        u.logger.ErrorWithContext(ctx, "Failed to save failed task event",
            u.logger.String("task_id", taskID),
            u.logger.String("task_type", taskType),
            u.logger.Error("err", err))
        return fmt.Errorf("save failed task event: %w", err)
    }
    
    u.tracing.TraceEvent(span, "Failed task event recorded successfully")
    u.tracing.RecordSpanAttributes(span, attribute.String("error.type", errorType))
    
    u.logger.InfoWithContext(ctx, "Failed task event recorded",
        u.logger.String("task_id", taskID),
        u.logger.String("task_type", taskType),
        u.logger.String("error_type", errorType),
        u.logger.Int("retry_count", retryCount))
    
    return nil
}

func (u *FailedTaskEventUseCase) GetErrorStatistics(ctx context.Context, startTime, endTime time.Time) (map[string]int, error) {
    ctx, span := u.tracing.StartSpan(ctx, "FailedTaskEventUseCase.GetErrorStatistics")
    defer u.tracing.SpanEnd(span)
    
    u.tracing.RecordSpanAttributes(span,
        attribute.String("time.start", startTime.Format(time.RFC3339)),
        attribute.String("time.end", endTime.Format(time.RFC3339)),
    )
    
    stats, err := u.repo.GetErrorStatistics(ctx, startTime, endTime)
    if err != nil {
        u.tracing.RecordSpanError(span, err)
        return nil, fmt.Errorf("get error statistics: %w", err)
    }
    
    totalErrors := 0
    for _, count := range stats {
        totalErrors += count
    }
    
    u.tracing.RecordSpanAttributes(span, attribute.Int("statistics.total_errors", totalErrors))
    u.tracing.TraceEvent(span, "Error statistics retrieved successfully")
    
    return stats, nil
}

func (u *FailedTaskEventUseCase) CleanupOldEvents(ctx context.Context, retentionDays int) error {
    ctx, span := u.tracing.StartSpan(ctx, "FailedTaskEventUseCase.CleanupOldEvents")
    defer u.tracing.SpanEnd(span)
    
    u.tracing.RecordSpanAttributes(span, attribute.Int("retention.days", retentionDays))
    
    cutoffTime := time.Now().AddDate(0, 0, -retentionDays)
    deletedCount, err := u.repo.DeleteOldEvents(ctx, cutoffTime)
    if err != nil {
        u.tracing.RecordSpanError(span, err)
        u.tracing.RecordSpanStatus(span, codes.Error, fmt.Sprintf("Cleanup failed: %v", err))
        return fmt.Errorf("cleanup old events: %w", err)
    }
    
    u.tracing.RecordSpanAttributes(span, attribute.Int64("cleanup.deleted_count", deletedCount))
    u.tracing.TraceEvent(span, "Old events cleanup completed")
    
    u.logger.InfoWithContext(ctx, "Cleaned up old failed task events",
        u.logger.Int64("deleted_count", deletedCount),
        u.logger.Int("retention_days", retentionDays),
        u.logger.String("cutoff_time", cutoffTime.Format(time.RFC3339)))
    
    return nil
}

// classifyError 根據錯誤訊息分類錯誤類型
func (u *FailedTaskEventUseCase) classifyError(errorMessage string) string {
    errorLower := strings.ToLower(errorMessage)
    
    switch {
    case strings.Contains(errorLower, "timeout"):
        return "TIMEOUT"
    case strings.Contains(errorLower, "connection") || strings.Contains(errorLower, "connect"):
        return "CONNECTION"
    case strings.Contains(errorLower, "unmarshal") || strings.Contains(errorLower, "marshal") || strings.Contains(errorLower, "json"):
        return "DATA_FORMAT"
    case strings.Contains(errorLower, "validation") || strings.Contains(errorLower, "invalid"):
        return "VALIDATION"
    case strings.Contains(errorLower, "database") || strings.Contains(errorLower, "sql"):
        return "DATABASE"
    case strings.Contains(errorLower, "redis"):
        return "REDIS"
    case strings.Contains(errorLower, "context"):
        return "CONTEXT"
    default:
        return "UNKNOWN"
    }
}

// internal/infrastructure/cache/redis/cleanup_service.go
package redis

import (
    "context"
    "fmt"
    "time"
    
    "github.com/redis/go-redis/v9"
    "github.com/jvdiamondtech/ms-identity-cat/internal/domain/ports/outbound/infrastructure"
    "go.opentelemetry.io/otel/attribute"
)

type CleanupService struct {
    client  *redis.Client
    logger  infrastructure.Logger
    tracing infrastructure.TracingService
}

func NewCleanupService(
    client *redis.Client,
    logger infrastructure.Logger,
    tracing infrastructure.TracingService,
) *CleanupService {
    return &CleanupService{
        client:  client,
        logger:  logger,
        tracing: tracing,
    }
}

func (s *CleanupService) CleanupFailedTaskData(ctx context.Context) (int, error) {
    ctx, span := s.tracing.StartSpan(ctx, "CleanupService.CleanupFailedTaskData")
    defer s.tracing.SpanEnd(span)
    
    startTime := time.Now()
    
    // 清理失敗任務相關的Redis key
    patterns := []string{
        "asynq:default:failed:*",
        "asynq:default:dead:*", 
        "asynq:critical:failed:*",
        "asynq:critical:dead:*",
        "asynq:default:completed:*",
        "asynq:default:processed:*",
        "asynq:critical:completed:*",
        "asynq:critical:processed:*",
    }
    
    totalDeleted := 0
    cutoffTime := time.Now().Add(-2 * time.Hour).Unix() // 保留2小時內的資料
    
    s.tracing.RecordSpanAttributes(span,
        attribute.Int64("cleanup.cutoff_time", cutoffTime),
        attribute.Int("cleanup.pattern_count", len(patterns)),
    )
    
    for _, pattern := range patterns {
        patternStartTime := time.Now()
        
        keys, err := s.client.Keys(ctx, pattern).Result()
        if err != nil {
            s.logger.ErrorWithContext(ctx, "Failed to get keys for cleanup",
                s.logger.String("pattern", pattern),
                s.logger.Error("err", err))
            continue
        }
        
        deletedCount := 0
        for _, key := range keys {
            // 檢查key的TTL
            ttl, err := s.client.TTL(ctx, key).Result()
            if err != nil {
                continue
            }
            
            // 刪除沒有TTL(-1)或已過期的key，以及超過保留時間的key
            if ttl == -1 || ttl < 0 || ttl < 2*time.Hour {
                if err := s.client.Del(ctx, key).Err(); err == nil {
                    deletedCount++
                    totalDeleted++
                }
            }
        }
        
        patternDuration := time.Since(patternStartTime)
        
        s.logger.InfoWithContext(ctx, "Cleaned up failed task Redis keys",
            s.logger.String("pattern", pattern),
            s.logger.Int("total_keys", len(keys)),
            s.logger.Int("deleted_count", deletedCount),
            s.logger.Duration("duration", patternDuration))
    }
    
    duration := time.Since(startTime)
    
    s.tracing.RecordSpanAttributes(span,
        attribute.Int("cleanup.total_deleted", totalDeleted),
        attribute.Int64("cleanup.duration_ms", duration.Milliseconds()),
    )
    s.tracing.TraceEvent(span, "Redis failed task data cleanup completed successfully")
    
    s.logger.InfoWithContext(ctx, "Redis failed task data cleanup completed",
        s.logger.Int("total_deleted", totalDeleted),
        s.logger.Duration("duration", duration))
    
    return totalDeleted, nil
}

func NewWorkerServer(
    cfg *config.Config, 
    logger infrastructure.Logger, 
    failedTaskUseCase inbound.FailedTaskEventUseCase,
) (*asynq.Server, error) {
    redisAddr := fmt.Sprintf("%s:%d", cfg.Redis.Domain, cfg.Redis.Port)

    redisOpt := asynq.RedisClientOpt{
        Addr:     redisAddr,
        Password: cfg.Redis.Password,
        DB:       cfg.Redis.DB,
    }

    // 設置服務器配置
    concurrency := 10
    queues := map[string]int{
        "default":  5,
        "critical": 10,
    }

    server := asynq.NewServer(
        redisOpt,
        asynq.Config{
            Concurrency: concurrency,
            Queues:      queues,
            RetryDelayFunc: func(n int, err error, task *asynq.Task) time.Duration {
                // 記錄重試事件到資料庫
                if task != nil {
                    ctx := context.Background()
                    taskID := getTaskIDFromTask(task)
                    queueName := extractQueueName(task)
                    
                    // 提取追蹤信息
                    var traceID, spanID *string
                    if span := trace.SpanFromContext(ctx); span.SpanContext().IsValid() {
                        traceIDStr := span.SpanContext().TraceID().String()
                        spanIDStr := span.SpanContext().SpanID().String()
                        traceID = &traceIDStr
                        spanID = &spanIDStr
                    }
                    
                    // 記錄失敗任務事件 (重試階段)
                    if recErr := failedTaskUseCase.RecordFailedTask(
                        ctx, taskID, task.Type(), queueName, 
                        string(task.Payload()), err.Error(), n, traceID, spanID,
                    ); recErr != nil {
                        logger.ErrorLog("Failed to record failed task during retry",
                            logger.String("task_id", taskID),
                            logger.Error("err", recErr))
                    }
                }

                // 指數退避策略
                delay := time.Duration(n*n) * time.Second
                maxDelay := 5 * time.Minute
                if delay > maxDelay {
                    delay = maxDelay
                }
                return delay
            },
            ErrorHandler: asynq.ErrorHandlerFunc(
                func(ctx context.Context, task *asynq.Task, err error) {
                    taskID := getTaskIDFromTask(task)
                    queueName := extractQueueName(task)
                    
                    // 提取追蹤信息
                    var traceID, spanID *string
                    if span := trace.SpanFromContext(ctx); span.SpanContext().IsValid() {
                        traceIDStr := span.SpanContext().TraceID().String()
                        spanIDStr := span.SpanContext().SpanID().String()
                        traceID = &traceIDStr
                        spanID = &spanIDStr
                    }
                    
                    // 記錄最終失敗的任務
                    if recErr := failedTaskUseCase.RecordFailedTask(
                        ctx, taskID, task.Type(), queueName,
                        string(task.Payload()), err.Error(), -1, // -1 表示最終失敗
                        traceID, spanID,
                    ); recErr != nil {
                        logger.ErrorLog("Failed to record final failed task",
                            logger.String("task_id", taskID),
                            logger.Error("err", recErr))
                    }
                    
                    logger.ErrorLog("Task processing error (final failure)",
                        logger.String("task_id", taskID),
                        logger.String("type", task.Type()),
                        logger.String("queue", queueName),
                        logger.Error("err", err))
                },
            ),
        },
    )

    return server, nil
}

// 輔助函數
func getTaskIDFromTask(task *asynq.Task) string {
    if task == nil {
        return "unknown"
    }
    if w := task.ResultWriter(); w != nil {
        return w.TaskID()
    }
    return fmt.Sprintf("no-writer-%s", uuid.NewString())
}

func extractQueueName(task *asynq.Task) string {
    if task == nil {
        return "unknown"
    }
    return "default" // 可以根據實際需要改進
}

// 更新 cmd/worker/worker.go
func runWorker(cobraCmd *cobra.Command, args []string) {
    cfg := cmd.GetConfig()
    logger := cmd.GetLogger()

    rootCtx := context.Background()

    s, err := initializeServices(rootCtx, cfg, logger)
    if err != nil {
        logger.FatalWithContext(rootCtx, "Failed to initialize services", logger.Error("error", err))
    }
    defer s.cleanup(rootCtx, logger)

    mux := asynq.NewServeMux()

    logger.InfoWithContext(rootCtx, "Starting worker...",
        logger.String("redis_domain", cfg.Redis.Domain),
        logger.Int("redis_port", cfg.Redis.Port))

    s.components.Handler.RegisterHandlers(mux)

    // 啟動簡化的定時清理任務
    cleanupCtx, cleanupCancel := context.WithCancel(rootCtx)
    defer cleanupCancel()
    
    go s.startSimpleCleanupScheduler(cleanupCtx, logger)

    go func() {
        if err := s.components.Server.Start(mux); err != nil {
            if !errors.Is(err, asynq.ErrServerClosed) {
                logger.FatalWithContext(rootCtx, "Failed to start worker server", logger.Error("err", err))
            }
        }
    }()

    quit := make(chan os.Signal, 1)
    signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
    <-quit

    logger.InfoWithContext(rootCtx, "Shutting down worker...")
    cleanupCancel()

    shutdownCtx, cancel := context.WithTimeout(rootCtx, 10*time.Second)
    defer cancel()

    done := make(chan struct{})
    go func() {
        s.components.Server.Shutdown()
        close(done)
    }()

    select {
    case <-done:
        logger.InfoWithContext(rootCtx, "Worker service exited gracefully")
    case <-shutdownCtx.Done():
        logger.WarnWithContext(rootCtx, "Worker service forced to shutdown after 10 seconds")
    }

    logger.InfoWithContext(rootCtx, "Worker exited")
}

// startSimpleCleanupScheduler 啟動簡化的清理排程器
func (s *services) startSimpleCleanupScheduler(ctx context.Context, logger infrastructure.Logger) {
    // 資料庫清理：每天執行一次
    dbCleanupTicker := time.NewTicker(24 * time.Hour)
    defer dbCleanupTicker.Stop()
    
    // Redis清理：每6小時執行一次
    redisCleanupTicker := time.NewTicker(6 * time.Hour)
    defer redisCleanupTicker.Stop()
    
    logger.InfoWithContext(ctx, "Simple cleanup scheduler started",
        logger.String("db_cleanup_interval", "24h"),
        logger.String("redis_cleanup_interval", "6h"))
    
    // 立即執行一次Redis清理
    go func() {
        if err := s.performSimpleRedisCleanup(ctx, logger); err != nil {
            logger.ErrorWithContext(ctx, "Initial Redis cleanup failed", logger.Error("err", err))
        }
    }()
    
    for {
        select {
        case <-dbCleanupTicker.C:
            go s.performDatabaseCleanup(ctx, logger)
            
        case <-redisCleanupTicker.C:
            go s.performSimpleRedisCleanup(ctx, logger)
            
        case <-ctx.Done():
            logger.InfoWithContext(ctx, "Simple cleanup scheduler stopped")
            return
        }
    }
}

// performDatabaseCleanup 執行資料庫清理
func (s *services) performDatabaseCleanup(ctx context.Context, logger infrastructure.Logger) {
    startTime := time.Now()
    
    logger.InfoWithContext(ctx, "Starting database cleanup...")
    
    // 清理30天前的錯誤事件
    retentionDays := 30
    if err := s.components.FailedTaskUseCase.CleanupOldEvents(ctx, retentionDays); err != nil {
        logger.ErrorWithContext(ctx, "Database cleanup failed", logger.Error("err", err))
        return
    }
    
    duration := time.Since(startTime)
    logger.InfoWithContext(ctx, "Database cleanup completed",
        logger.Int("retention_days", retentionDays),
        logger.Duration("duration", duration))
}

// performSimpleRedisCleanup 執行Redis清理
func (s *services) performSimpleRedisCleanup(ctx context.Context, logger infrastructure.Logger) {
    startTime := time.Now()
    
    logger.InfoWithContext(ctx, "Starting Redis cleanup...")
    
    // 清理失敗任務資料
    deletedCount, err := s.components.CleanupService.CleanupFailedTaskData(ctx)
    if err != nil {
        logger.ErrorWithContext(ctx, "Redis cleanup failed", logger.Error("err", err))
        return
    }
    
    duration := time.Since(startTime)
    logger.InfoWithContext(ctx, "Redis cleanup completed",
        logger.Int("deleted_count", deletedCount),
        logger.Duration("duration", duration))
}
